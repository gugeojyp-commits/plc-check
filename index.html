function doPost(e) { const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("명단"); const data = sheet.getDataRange().getValues(); const params = JSON.parse(e.postData.contents); const action = params.action; const name = params.name.trim(); // 1. 이수 여부 및 참여 시간(G열)/분(H열) 조회 if (action === "check") { for (let i = 1; i < data.length; i++) { const sheetName = String(data[i][0]).trim(); if (sheetName === name) { const status = String(data[i][5]).trim(); // F열: 이수여부 ('이수' 또는 '미이수') const hours = data[i][6] !== "" ? data[i][6] : 0; // G열: 시간 const minutes = data[i][7] !== "" ? data[i][7] : 0; // H열: 분 if (status === "이수") { return createResponse({ status: "found", hours: hours, minutes: minutes, alreadySubmitted: data[i][3] !== "" // D열: 나이스번호 기제출 여부 }); } else { return createResponse({ status: "not_completed", hours: hours, minutes: minutes }); } } } return createResponse({ status: "not_found" }); } // 2. 생년월일, 직급, 나이스 개인번호 저장 if (action === "submit") { const birth = String(params.birth).trim(); const position = params.position.trim(); const neisId = params.neisId.trim().toUpperCase(); for (let i = 1; i < data.length; i++) { const sheetName = String(data[i][0]).trim(); const status = String(data[i][5]).trim(); if (sheetName === name && status === "이수") { sheet.getRange(i + 1, 2).setValue("'" + birth); // B열: 생년월일 sheet.getRange(i + 1, 3).setValue(position); // C열: 직급 sheet.getRange(i + 1, 4).setValue(neisId); // D열: 나이스 개인번호 sheet.getRange(i + 1, 5).setValue(new Date()); // E열: 제출일시 return createResponse({ status: "success" }); } } return createResponse({ status: "error" }); } } function createResponse(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
